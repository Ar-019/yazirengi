@Composable
fun ButtonExamples(modifier: Modifier = Modifier) {
    var arkaplanrengi by remember {mutableStateOf(Color.White)}
    Box(
        modifier = modifier
        .fillMaxSize()
        .background(arkaplanrengi)
    )
    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(48.dp),
        verticalArrangement = Arrangement.spacedBy(24.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
    ) {
        Text("Filled button:")
        FilledButtonExample(onClick = { arkaplanrengi = Color.Red })
        Text("Filled tonal button:")
        FilledTonalButtonExample(onClick = { arkaplanrengi = Color.Blue })
        Text("Elevated button:")
        ElevatedButtonExample(onClick = { arkaplanrengi = Color.Blue })
        Text("Outlined button:")
        OutlinedButtonExample(onClick = { arkaplanrengi = Color.Blue })
        Text("Text button")
        TextButtonExample(onClick = { arkaplanrengi = Color.Black })
    }
}
